## Installation Commands

# Install Forge dependencies
cd forge && npm install

# Install Mistral Chat Frontend dependencies
cd mistral-chat/frontend && npm install

# Install Mistral Chat Backend dependencies
cd mistral-chat/backend && npm install

## Development Commands

# Start Forge (full stack)
cd forge && npm run dev

# Start Mistral Chat Backend
cd mistral-chat/backend && npm run dev

# Start Mistral Chat Frontend
cd mistral-chat/frontend && npm run dev

## Production Build Commands

# Build Forge
cd forge && npm run build

# Build Mistral Chat Frontend
cd mistral-chat/frontend && npm run build
